& {
    $ErrorActionPreference = 'Stop'
    $expected = '8C47689AE933DD366930339AC03195C3B5A7972341DA4B8567B9EF9862E5FC7B'
    $entry = 'Botrytis_Experiment_Framework/scripts/Invoke-SuiteIntegrada.ps1'
    $downloadValue = (Get-ItemProperty -LiteralPath 'HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\User Shell Folders' -Name '{374DE290-123F-4565-9164-39C4925E467B}' -ErrorAction SilentlyContinue).'{374DE290-123F-4565-9164-39C4925E467B}'
    $downloads = if ($downloadValue) { [Environment]::ExpandEnvironmentVariables($downloadValue) } else { Join-Path $env:USERPROFILE 'Downloads' }
    $archive = Get-ChildItem -LiteralPath $downloads -Filter 'Botrytis_Suite_Integrada_Botrytest*.zip' -File |
        Where-Object { (Get-FileHash -LiteralPath $_.FullName -Algorithm SHA256).Hash -eq $expected } |
        Select-Object -First 1
    if (-not $archive) { throw 'ZIP Botrytis_Suite_Integrada_Botrytest verificado no encontrado en Descargas.' }
    $base = Join-Path $env:USERPROFILE 'Documents/s2_no_borrar/HybridSuite/install'
    New-Item -ItemType Directory -Path $base -Force | Out-Null
    $lock = [IO.File]::Open((Join-Path $base 'launcher.lock'), 'OpenOrCreate', 'ReadWrite', 'None')
    try {
        Add-Type -AssemblyName System.IO.Compression.FileSystem
        $version = Join-Path $base $expected.Substring(0, 16)
        $valid = Test-Path -LiteralPath (Join-Path $version $entry) -PathType Leaf
        if ($valid) {
            $package = [IO.Compression.ZipFile]::OpenRead($archive.FullName)
            $sha = [Security.Cryptography.SHA256]::Create()
            try {
                foreach ($file in $package.Entries) {
                    if (-not $file.Name) { continue }
                    $installed = Join-Path $version $file.FullName
                    if (-not (Test-Path -LiteralPath $installed -PathType Leaf)) { $valid = $false; break }
                    $stream = $file.Open()
                    try { $hash = [BitConverter]::ToString($sha.ComputeHash($stream)).Replace('-', '') } finally { $stream.Dispose() }
                    if ((Get-FileHash -LiteralPath $installed -Algorithm SHA256).Hash -ne $hash) { $valid = $false; break }
                }
            } finally { $sha.Dispose(); $package.Dispose() }
        }
        if (-not $valid) {
            $staging = Join-Path $base ('staging-' + [Guid]::NewGuid().ToString('N'))
            try {
                Expand-Archive -LiteralPath $archive.FullName -DestinationPath $staging
                if (Test-Path -LiteralPath $version) {
                    Move-Item -LiteralPath $version -Destination (Join-Path $base ('preserved-' + [Guid]::NewGuid().ToString('N')))
                }
                Move-Item -LiteralPath $staging -Destination $version
            } finally {
                if (Test-Path -LiteralPath $staging) { Remove-Item -LiteralPath $staging -Recurse -Force }
            }
        }
        $launch = [ScriptBlock]::Create([IO.File]::ReadAllText((Join-Path $version $entry)))
        & $launch -Mode 'Today' -Downloads $downloads -ProjectRoot (Join-Path $version 'Botrytis_Experiment_Framework')
    } finally { $lock.Dispose() }
}
